* means required. Unless specified, authenticated endpoints should be verified with backend.

AUTHENTICATION
POST /auth/register — first_name*, last_name*, email*, password*, password_confirmation, date_of_birth, gender, height, height_unit(cm|in), current_weight, weight_unit(kg|lb), address, timezone, activity_level(sedentary|light|moderate|high|very_high) — 201; 422 validation; 429 throttle
POST /auth/login — email*, password* — 200; 401 invalid credentials
GET /auth/me — 200
POST /auth/logout — 200

GOALS
GET /goals — 200
POST /goals — goal_type*, name*, description, target_value*, unit*, period*(day|week|month), start_date*, end_date — 201
GET /goals/{goal} — 200
PUT /goals/{goal} — goal_type, name, description, target_value, unit, period, start_date, end_date, active — 200; 422 empty body
DELETE /goals/{goal} — 200
GET /goals/{goal}/progress — 200
POST /goals/{goal}/progress — date*, value*, completed* — 200

STREAKS
GET /streaks — 200
GET /streaks/{type} — 200

FASTING
GET /fasting — query status(ongoing|completed|failed), from, to — 200
POST /fasting — fasting_type*, planned_hours*(1-48), started_at, notes — 201
GET /fasting/active — 200
GET /fasting/summary — 200
GET /fasting/{fasting} — 200
PUT /fasting/{fasting} — fasting_type, planned_hours, notes — 200; 422 empty body
POST /fasting/{fasting}/end — ended_at — 200
DELETE /fasting/{fasting} — 200

PROFILE
GET /profile — 200
PUT /profile — first_name, last_name, date_of_birth, gender, height, height_unit, current_weight, weight_unit, body_fat_percentage, address, timezone, activity_level — 200; 422 empty body

WEIGHT
GET /weight-logs — query from, to, limit(1-365) — 200
POST /weight-logs — weight*(1-700), unit(kg|lb), logged_at, source(manual|health_platform|wearable), external_id, notes — 201
GET /weight-logs/latest — 200
GET /weight-logs/{weightLog} — 200
PUT /weight-logs/{weightLog} — weight, unit, logged_at, notes — 200; 422 empty body
DELETE /weight-logs/{weightLog} — 200

NUTRITION: FOODS
GET /foods — query q, brand, category_id, barcode, mine, sort(relevance|name|calories|recent), limit(1-100) — 200
POST /foods — name*, calories*, protein*, carbohydrates*, fat*, brand, category_id, barcode, base_unit(g|ml), base_amount, fiber, sugar, saturated_fat, sodium, grams_per_unit, ml_per_unit, serving_amount, source — 201
GET /food-categories — 200
GET /foods/barcode/{barcode} — 200 local; 201 created from database; 404 not found
GET /foods/{food} — 200
PUT /foods/{food} — create fields all optional — 200; 422 empty body
DELETE /foods/{food} — 200

NUTRITION: MEALS
GET /meals — query date, from, to, meal_type — 200
POST /meals — meal_type*, name, logged_at, notes, items[](food_id, name, quantity, unit, calories, protein, carbohydrates, fat, fiber) — 201 (+active_fasting)
GET /meals/{meal} — 200
PUT /meals/{meal} — meal_type, name, logged_at, notes — 200; 422 empty body
DELETE /meals/{meal} — 200
POST /meals/{meal}/items — food_id OR name*, quantity*, unit, calories, protein, carbohydrates, fat, fiber, sugar, saturated_fat, sodium — 201
PUT /meals/{meal}/items/{item} — same as add; quantity required if food_id changes — 200; 422 empty body
DELETE /meals/{meal}/items/{item} — 200

NUTRITION: DAY/LOG/RECOGNITION
GET /nutrition/day — query date — 200
POST /nutrition/log — meal_type*, quantity*, name*(or food_id), calories*(or food_id), unit, logged_at, protein, carbohydrates, fat, fiber — 201 (+active_fasting)
POST /nutrition/recognize — type*(photo|voice|text), text, image_base64, meal_type — 200 draft only, not persisted

NUTRITION: RECIPES
GET /recipes — 200
POST /recipes — name*, items*[](food_id|name, quantity, unit, calories, protein, carbohydrates, fat, fiber), notes, servings(1-100) — 201
GET /recipes/{recipe} — 200
PUT /recipes/{recipe} — name, notes, servings, items[] — 200; 422 empty body
DELETE /recipes/{recipe} — 200

NUTRITION: FAVORITES
GET /favorites — query type(food|meal|meditation) — 200
POST /favorites — type*(food|meal|meditation), id* — 201
DELETE /favorites/{favorite} — 200

ENERGY
GET /energy/balance — query date — 200
GET /energy/bmr — 200
POST /energy/targets/calculate — direction(maintain|lose|gain) — 200

MEDITATION: CONTENT

GET /meditation/{meditation} — 200
GET /meditation/categories — 200

MEDITATION: HOME AND STATS
GET /meditation/home — 200
GET /meditation/stats — 200

MEDITATION: SESSIONS
GET /meditation/sessions — optional query status(active|paused|completed|cancelled), type, from, to — 200
GET /meditation/sessions/active — 200, session:null if none
POST /meditation/sessions — planned_minutes*; type* if no meditation_id, meditation_id optional, started_at, notes — 201
GET /meditation/sessions/{session} — 200
PUT /meditation/sessions/{session} — notes(max 1000 characters) — 200
POST /meditation/sessions/{session}/pause — 200
POST /meditation/sessions/{session}/resume — 200
POST /meditation/sessions/{session}/complete — optional ended_at(<=now) — 200
POST /meditation/sessions/{session}/cancel — 200
DELETE /meditation/sessions/{session} — 200

MEDITATION: GOALS
GET /meditation/goals — 200
POST /meditation/goals — name, target_value, unit, period(day|week|month), start_date*, description, end_date — 201
GET /meditation/goals/{goal} — 200
PUT /meditation/goals/{goal} — name, description, target_value, unit, period, start_date, end_date, active — 200
DELETE /meditation/goals/{goal} — 200
GET /meditation/goals/{goal}/progress — 200
POST /meditation/goals/{goal}/progress — date*, value* — 201

MEDITATION: REMINDERS
GET /meditation/reminders — 200
POST /meditation/reminders — time*(HH:MM), days_of_week(e.g. "0,1,2"), label, enabled — 201
PUT /meditation/reminders/{reminder} — time, days_of_week, label, enabled — 200
DELETE /meditation/reminders/{reminder} — 200

- Endpoint availability and contracts must be checked against the live backend.
- Do not infer that absent modules like activity/yoga/training are implemented.
- GET /foods query uses q per doc; validate against observed backend handling of search.
